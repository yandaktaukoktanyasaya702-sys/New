local function validKey(name)
    if type(name) ~= "string" or name == "Unknown" then return false end
    local ok, value = pcall(function() return Enum.KeyCode[name] end)
    return ok and value ~= nil
end

local function mergeSettings(data)
    if type(data) ~= "table" then return end
    for key, base in pairs(defaults) do
        local value = data[key]
        if type(value) == type(base) then
            if type(value) == "number" then
                if value == value and math.abs(value) < math.huge then
                    CFG[key] = math.clamp(value, limits[key][1], limits[key][2])
                end
            elseif key == "theme" then
                if themes[value] then CFG[key] = value end
            elseif key == "aimPart" then
                if value == "Head" or value == "Chest" then CFG[key] = value end
            else CFG[key] = value end
        end
    end
end

local function make(className, props, parent)
    local object = Instance.new(className)
    if object:IsA("GuiObject") then object.BorderSizePixel = 0 end
    for key, value in pairs(props or {}) do object[key] = value end
    object.Parent = parent
    return object

local function label(parent, text, size, position, dims, token, font)

    return alive and playable() and not typing and UIS:GetFocusedTextBox() == nil and not (menuOpen and CFG.pauseOnMenu)

    return combatAllowed() and (not CFG.aimHold or rightHeld)

local function aimPoint(record, predict)
    local point = record.feet + Vector3.new(0, record.height * (CFG.aimPart == "Head" and 0.88 or 0.60), 0)
    if predict then
        local lead = record.velocity * CFG.prediction
        if lead.Magnitude > 1.5 then lead = lead.Unit * 1.5 end
        point = point + lead
    end
    return point
end
-- MC coordinates are projected relative to Controller.eye(). A uniform voxel
-- scale cancels in perspective, so no guessed studs-per-block multiplier is used.
local function project(camera, eye, point)
    return camera:WorldToViewportPoint(camera.CFrame.Position + point - eye)
end
local function screenDistance(camera, eye, point)
    local p, visible = project(camera, eye, point)
    if not visible or p.Z <= 0.01 then return math.huge end
    local viewport = camera.ViewportSize
    local dx, dy = p.X - viewport.X * 0.5, p.Y - viewport.Y * 0.5
    return math.sqrt(dx * dx + dy * dy)
end
local function usable(record, maxRange, now)
    return record ~= nil and now - record.seen < 0.3 and record.distance <= maxRange
        and (record.health == nil or record.health > 0)
end
local function snapshot(look)
    local state = Controller.state
    if not state or not vector(state.pos) then return nil end
    return {fall = state.fallDistance or 0, ground = state.onGround == true,
        sprint = state.sprinting == true, water = state.inWater == true, gliding = state.gliding == true,
        vel = state.vel or Vector3.zero, look = look, swap = false, pos = state.pos, predicted = true}
end
local function readIdentity(target)
    local entity = target.player
    local name = target.displayName or target.name
    local health = finite(target.health) and target.health or (finite(target.hp) and target.hp or nil)
    local maxHealth = finite(target.maxHealth) and target.maxHealth or nil
    if typeof(entity) == "Instance" and entity:IsA("Player") then
        name = entity.DisplayName
        local hp = entity:GetAttribute("Health")
        if health == nil and finite(hp) then health = hp end
        local max = entity:GetAttribute("MaxHealth")
        if maxHealth == nil and finite(max) then maxHealth = max end
    end
    -- Missing health remains unknown. Never manufacture a full health bar.
    return string.sub(tostring(name or target.id), 1, 32), health, maxHealth
end
local function scanTargets(now)
    local started = os.clock()
    lastScan = now
    table.clear(candidates)
    aimTarget, auraTarget, silentTarget = nil, nil, nil
    if not (CFG.esp or CFG.camLock or CFG.killaura or CFG.silentAim) then
        count = 0; table.clear(cache); scanCost = 0; return
    end
    local eye, camera = eyePosition(), workspace.CurrentCamera
    if not eye or not camera then count = 0; return end
    local ok, targets = pcall(World.targets)
    if not ok or type(targets) ~= "table" then
        count = 0; scanCost = (os.clock() - started) * 1000; return
    end
    local nearest, bestAim, bestSilent = math.huge, math.huge, math.huge
    local previousId = api.targetId
    local sticky
    local inspected = 0
    for _, target in pairs(targets) do
        inspected = inspected + 1
        if inspected > 512 then break end
        if type(target) == "table" and (type(target.id) == "string" or finite(target.id)) and vector(target.feet) then
            local record = cache[target.id]
            local fresh = record == nil
            if fresh then
                record = {id = target.id, velocity = Vector3.zero, seen = now, feet = target.feet}
                cache[target.id] = record
            end
            local elapsed = now - record.seen
            if not fresh and elapsed >= 0.01 and elapsed <= 0.3 then
                local velocity = (target.feet - record.feet) / elapsed
                if velocity.Magnitude < 60 then
                    record.velocity = record.velocity:Lerp(velocity, 0.4)
                else record.velocity = Vector3.zero end
            else record.velocity = Vector3.zero end
            record.feet, record.seen = target.feet, now
            record.height = finite(target.height) and math.clamp(target.height, 0.2, 8) or 1.8
            record.width = finite(target.width) and math.clamp(target.width, 0.1, 4) or 0.6
            record.name, record.health, record.maxHealth = readIdentity(target)
            record.distance = (target.feet + Vector3.new(0, record.height * 0.5, 0) - eye).Magnitude
            if record.health == nil or record.health > 0 then
                candidates[#candidates + 1] = record
                local point = aimPoint(record, false)
                local delta = point - eye
                local withinAngle = CFG.auraFov >= 360 or (delta.Magnitude > 0.001
                    and camera.CFrame.LookVector:Dot(delta.Unit) >= math.cos(math.rad(CFG.auraFov * 0.5)))
                if CFG.killaura and record.distance <= CFG.range and record.distance < nearest and withinAngle then
                    auraTarget, nearest = record, record.distance
                end
                if (CFG.camLock and record.distance <= CFG.aimRange) or (CFG.silentAim and record.distance <= CFG.silentRange) then
                    local distance = screenDistance(camera, eye, point)
                    record.screenDistance = distance
                    if distance <= CFG.aimFov then
                        if CFG.camLock and record.distance <= CFG.aimRange then
                            if record.id == previousId then sticky = record end
                            if distance < bestAim then aimTarget, bestAim = record, distance end
                        end
                        if CFG.silentAim and record.distance <= CFG.silentRange and distance < bestSilent then
                            silentTarget, bestSilent = record, distance
                        end
                    end
                end
            end
        end
    end
    if CFG.aimSticky and sticky and (not aimTarget or sticky.screenDistance <= bestAim + 30) then aimTarget = sticky end
    api.targetId = aimTarget and aimTarget.id or nil
    for id, record in pairs(cache) do
        if record.seen ~= now then cache[id] = nil end
    end
    count = #candidates
    if CFG.esp then table.sort(candidates, function(a, b) return a.distance < b.distance end) end
    scanCost = (os.clock() - started) * 1000
end
local function aimCamera(dt)
    if not alive or not CFG.camLock or not cameraAllowed() then return end
    local now, target = os.clock(), aimTarget
    if not usable(target, CFG.aimRange, now) then return end
    local eye = eyePosition()
    if not eye or not finite(Controller.yaw) or not finite(Controller.pitch) then return end
    local delta = aimPoint(target, true) - eye
    if delta.Magnitude < 0.001 then return end
    local yaw = math.atan2(-delta.X, -delta.Z)
    local pitch = math.atan2(delta.Y, math.sqrt(delta.X * delta.X + delta.Z * delta.Z))
    local alpha = 1 - math.exp(-math.clamp(dt, 0, 0.1) * (21 - CFG.camSmooth) * 1.2)
    local difference = (yaw - Controller.yaw + math.pi) % (2 * math.pi) - math.pi
    Controller.yaw = (Controller.yaw + difference * alpha + math.pi) % (2 * math.pi) - math.pi
    Controller.pitch = math.clamp(Controller.pitch + (pitch - Controller.pitch) * alpha, -math.pi / 2 + 0.001, math.pi / 2 - 0.001)
end
local function doKillaura(now)
    if not CFG.killaura or not combatAllowed() or totTransaction then return end
    if now - lastAttack < CFG.attackInterval or not usable(auraTarget, CFG.range, now) then return end
    local okMode, mode = pcall(Hands.gamemode)
    if okMode and mode == "spectator" then return end
    local okStrength, strength = pcall(Hands.strength)
    if not okStrength or not finite(strength) or strength < 0.95 then return end
    local eye = eyePosition()
    if not eye then return end
    local delta = aimPoint(auraTarget, false) - eye
    if delta.Magnitude < 0.001 or delta.Magnitude > CFG.range + 0.2 then return end
    local state = snapshot(delta.Unit)
    if not state then return end
    lastAttack = now -- Bound retries even if the transport throws.
    -- Use the captured original while silent aim is active, preserving aura selection.
    local send = (wrappedSend and Net.send == wrappedSend) and originalSend or Net.send
    local ok = pcall(send, {a = "attack", target = auraTarget.id, st = state})
    if ok then Hands.attackReset = Controller.tickCount; Hands.swing = now end
end

-- A scoped table-function wrapper, never a global namecall/raycast hook.
-- Only the supplied MC melee packet shape is supported.
syncSilent = function()
    if not CFG.silentAim then
        if wrappedSend and Net.send == wrappedSend then pcall(function() Net.send = originalSend end) end
        wrappedSend, originalSend = nil, nil
        return true
    end
    if wrappedSend then
        if Net.send == wrappedSend then return true end
        report("Net.send changed; silent aim disabled to preserve the newer wrapper.")
        CFG.silentAim = false; wrappedSend, originalSend = nil, nil
        return false
    end
    local base = Net.send
    local random = Random.new()
    local wrapper = function(packet, ...)
        if alive and CFG.silentAim and combatAllowed() and type(packet) == "table"
            and packet.a == "attack" and type(packet.st) == "table" and vector(packet.st.look)
            and usable(silentTarget, CFG.silentRange, os.clock()) and random:NextNumber(0, 100) < CFG.silentChance then
            local eye = eyePosition()
            if eye then
                local delta = aimPoint(silentTarget, true) - eye
                if delta.Magnitude > 0.001 and delta.Magnitude <= CFG.silentRange + 0.2 then
                    local copy = table.clone(packet)
                    copy.st = table.clone(packet.st)
                    copy.target, copy.st.look = silentTarget.id, delta.Unit
                    return base(copy, ...)
                end
            end
        end
        return base(packet, ...)
    end
    local ok = pcall(function() Net.send = wrapper end)
    if not ok or Net.send ~= wrapper then
        CFG.silentAim = false; report("Silent aim unavailable: MC.Net.send is not writable.")
        return false
    end
    originalSend, wrappedSend = base, wrapper
    return true
end

-- Inventory work is a heartbeat state machine: no spawned retry threads.
local TOTEM_ID = "totem_of_undying"
local function itemId(item) return type(item) == "table" and item.id or nil end
local function clickSlot(slot) return pcall(Net.send, {a = "inv", op = "click", slot = slot, right = false, shift = false}) end
local function finishTransaction()
    local transaction, inv = totTransaction, Hands.inv
    if transaction and inv and inv.cursor then
        local owned = inv.cursor == transaction.ownedCursor
            or (transaction.phase == "pickup" and itemId(inv.cursor) == TOTEM_ID)
            or (transaction.phase == "place" and transaction.previousOffhand ~= nil and inv.cursor == transaction.previousOffhand)
        if owned then clickSlot(transaction.slot) end
    end
    totTransaction = nil
end
local function ensureTotem(now)
    local inv = Hands.inv
    if totTransaction then
        local transaction = totTransaction
        if not inv then totTransaction = nil; return end
        if now > transaction.deadline then finishTransaction(); return end
        if now < transaction.next then return end
        if transaction.phase == "pickup" then
            if itemId(inv.cursor) ~= TOTEM_ID then return end
            transaction.ownedCursor = inv.cursor
            if not clickSlot(10) then finishTransaction(); return end
            transaction.phase, transaction.next = "place", now + 0.12
        elseif transaction.phase == "place" then
            if itemId(inv.offhand) ~= TOTEM_ID then return end
            if inv.cursor then
                transaction.ownedCursor = inv.cursor
                clickSlot(transaction.slot)
            end
            totTransaction = nil
        end
        return
    end
    if not CFG.autoTotem or not combatAllowed() or not inv or now - lastTotem < 0.2 then return end
    lastTotem = now
    if itemId(inv.offhand) == TOTEM_ID or inv.cursor then return end
    local hp = player:GetAttribute("Health")
    if inv.offhand and (not finite(hp) or hp > CFG.totemHealth) then return end
    for slot = 1, 9 do
        if inv.hotbar and itemId(inv.hotbar[slot]) == TOTEM_ID then
            if inv.selected ~= slot then pcall(Hands.select, slot); return end
            if type(Hands.stopUsing) == "function" then pcall(Hands.stopUsing, false) end
            local oldOffhand, current = inv.offhand, inv.hotbar[slot]
            inv.hotbar[slot], inv.offhand = oldOffhand, current
            if not pcall(Net.send, {a = "swap"}) then
                inv.hotbar[slot], inv.offhand = current, oldOffhand
            else Hands.attackReset = Controller.tickCount end
            return
        end
    end
    local selected
    for index, item in pairs(inv.inventory or {}) do
        if type(index) == "number" and index >= 1 and index % 1 == 0 and itemId(item) == TOTEM_ID
            and (not selected or index < selected) then selected = index end
    end
    if selected then
        local slot = selected + 10
        if clickSlot(slot) then
            totTransaction = {slot = slot, phase = "pickup", previousOffhand = inv.offhand, next = now + 0.12, deadline = now + 1.2}
        end
    end
end

-- ESP objects are pooled by display slot. No instance creation in the render loop.
local espPool = {}
local function buildEspSlot()
    local root = make("Frame", {BackgroundTransparency = 1, Visible = false, Size = UDim2.fromOffset(1, 1)}, espLayer)
    local box = make("Frame", {BackgroundTransparency = 1, Size = UDim2.fromScale(1, 1)}, root)
    local stroke = make("UIStroke", {Thickness = 1.2, ApplyStrokeMode = Enum.ApplyStrokeMode.Border}, box)
    local segments = {}
    for index = 1, 8 do
        segments[index] = make("Frame", {}, root)
    end
    local name = make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.new(0.5, -90, 0, -19),
        Size = UDim2.fromOffset(180, 17), Font = Enum.Font.GothamMedium, TextSize = 11,
        TextColor3 = Color3.new(1, 1, 1), TextStrokeTransparency = 0.3}, root)
    local details = make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.new(0.5, -85, 1, 3),
        Size = UDim2.fromOffset(170, 16), Font = Enum.Font.Code, TextSize = 10,
        TextColor3 = Color3.new(1, 1, 1), TextStrokeTransparency = 0.3}, root)
    local healthBg = make("Frame", {Position = UDim2.fromOffset(-6, 0), Size = UDim2.new(0, 3, 1, 0), BackgroundColor3 = Color3.fromRGB(25, 25, 28)}, root)
    local healthFill = make("Frame", {AnchorPoint = Vector2.new(0, 1), Position = UDim2.fromScale(0, 1), Size = UDim2.fromScale(1, 1)}, healthBg)
    local tracer = make("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), Visible = false}, espLayer)
    return {root = root, box = box, stroke = stroke, segments = segments, name = name,
        details = details, healthBg = healthBg, healthFill = healthFill, tracer = tracer, textAt = 0}
end
local function resizePool()
    local limit = CFG.esp and math.floor(CFG.espLimit) or 0
    while #espPool < limit do espPool[#espPool + 1] = buildEspSlot() end
    while #espPool > limit do
        local slot = table.remove(espPool); slot.root:Destroy(); slot.tracer:Destroy()
    end
end
local function hideEsp()
    for _, slot in ipairs(espPool) do slot.root.Visible = false; slot.tracer.Visible = false end
end
local function bounds(camera, eye, record)
    local top = project(camera, eye, record.feet + Vector3.new(0, record.height, 0))
    local bottom = project(camera, eye, record.feet)
    if top.Z <= 0.05 or bottom.Z <= 0.05 then return nil end
    local centre = (top + bottom) * 0.5
    local side = project(camera, eye, record.feet + Vector3.new(0, record.height * 0.5, 0) + camera.CFrame.RightVector * (record.width * 0.5))
    local halfWidth = math.max(2, math.abs(side.X - centre.X))
    local left, right = math.min(top.X, bottom.X) - halfWidth, math.max(top.X, bottom.X) + halfWidth
    local upper, lower = math.min(top.Y, bottom.Y), math.max(top.Y, bottom.Y)
    local viewport = camera.ViewportSize
    if right < 0 or left > viewport.X or lower < 0 or upper > viewport.Y or lower - upper < 2 then return nil end
    if right - left > viewport.X * 2 or lower - upper > viewport.Y * 2 then return nil end
    return left, upper, right - left, lower - upper
end
local function drawEsp(now)
    local started = os.clock()
    local camera, eye = workspace.CurrentCamera, eyePosition()
    if not CFG.esp or not camera or not eye or now - lastScan > 0.3 then hideEsp(); return end
    local used = 0
    for _, record in ipairs(candidates) do
        if used >= #espPool then break end
        if usable(record, CFG.espRange, now) then
            local x, y, width, height = bounds(camera, eye, record)
            if x then
                used = used + 1
                local slot = espPool[used]
                slot.root.Visible = true
                slot.root.Position = UDim2.fromOffset(math.floor(x), math.floor(y))
                slot.root.Size = UDim2.fromOffset(math.floor(width), math.floor(height))
                local tint = (record == aimTarget or record == auraTarget or record == silentTarget) and theme.text or theme.accent
                local fade = CFG.espFade and math.clamp(record.distance / CFG.espRange * 0.65, 0, 0.65) or 0
                slot.box.Visible = CFG.espBoxes and not CFG.espCorners
                slot.stroke.Color, slot.stroke.Transparency = tint, fade
                for index, segment in ipairs(slot.segments) do
                    segment.Visible = CFG.espBoxes and CFG.espCorners
                    if segment.Visible then
                        local corner = math.floor((index - 1) / 2)
                        local rightSide, bottomSide = corner % 2 == 1, corner >= 2
                        segment.AnchorPoint = Vector2.new(rightSide and 1 or 0, bottomSide and 1 or 0)
                        segment.Position = UDim2.fromScale(rightSide and 1 or 0, bottomSide and 1 or 0)
                        segment.Size = index % 2 == 1 and UDim2.new(0.25, 0, 0, 1) or UDim2.new(0, 1, 0.22, 0)
                        segment.BackgroundColor3, segment.BackgroundTransparency = tint, fade
                    end
                end
                slot.name.Visible = CFG.espNames
                slot.name.TextTransparency, slot.details.TextTransparency = fade, fade
                local validHealth = CFG.espHealth and finite(record.health)
                local validBar = validHealth and finite(record.maxHealth) and record.maxHealth > 0
                slot.healthBg.Visible = validBar == true
                if validBar then
                    local ratio = math.clamp(record.health / record.maxHealth, 0, 1)
                    slot.healthFill.Size = UDim2.fromScale(1, ratio)
                    slot.healthFill.BackgroundColor3 = Color3.fromHSV(ratio * 0.33, 0.7, 0.95)
                end
                slot.details.Visible = CFG.espDistance or validHealth == true
                if slot.id ~= record.id or now - slot.textAt >= 0.2 then
                    slot.id, slot.textAt = record.id, now
                    slot.name.Text = record.name
                    local distance = CFG.espDistance and string.format("%.1f blocks", record.distance) or ""
                    slot.details.Text = distance .. (validHealth and string.format("%s%.0f HP", distance ~= "" and "  /  " or "", record.health) or "")
                end
                slot.tracer.Visible = CFG.espTracers
                if CFG.espTracers then
                    local origin = Vector2.new(camera.ViewportSize.X * 0.5, camera.ViewportSize.Y - 12)
                    local endpoint = Vector2.new(x + width * 0.5, y + height)
                    local delta = endpoint - origin
                    slot.tracer.Position = UDim2.fromOffset((origin.X + endpoint.X) * 0.5, (origin.Y + endpoint.Y) * 0.5)
                    slot.tracer.Size = UDim2.fromOffset(delta.Magnitude, 1)
                    slot.tracer.Rotation = math.deg(math.atan2(delta.Y, delta.X))
                    slot.tracer.BackgroundColor3, slot.tracer.BackgroundTransparency = tint, math.max(0.3, fade)
                end
            end
        end
    end
    for index = used + 1, #espPool do espPool[index].root.Visible = false; espPool[index].tracer.Visible = false end
    espCost = (os.clock() - started) * 1000
end

local function fitWindow()
    local camera = workspace.CurrentCamera
    if not camera then return end
    local view = camera.ViewportSize
    scale.Scale = math.max(0.2, math.min(CFG.uiScale, (view.X - 24) / 900, (view.Y - 48) / 596))
    local position = window.AbsolutePosition
    window.Position = UDim2.fromOffset(math.clamp(position.X, 8, math.max(8, view.X - 900 * scale.Scale - 8)),
        math.clamp(position.Y, 8, math.max(8, view.Y - 596 * scale.Scale - 8)))
end

setOption = function(key, value)
    if not alive or defaults[key] == nil then return end
    mergeSettings({[key] = value})
    if key == "theme" then applyTheme() end
    if key == "uiScale" then fitWindow() end
    if key == "silentAim" then syncSilent() end
    if key == "esp" or key == "espLimit" then resizePool() end
    if key == "camLock" and not CFG.camLock then aimTarget = nil; api.targetId = nil end
    if key == "killaura" and not CFG.killaura then auraTarget = nil end
    if key == "autoTotem" and not CFG.autoTotem then finishTransaction() end
    for _, slot in ipairs(espPool) do slot.textAt = 0 end
    refreshAll()
end
local function settingsSnapshot()
    local values = {}
    for key in pairs(defaults) do values[key] = CFG[key] end
    return {version = 2, settings = values, binds = table.clone(binds)}
end
local function loadProfile(data)
    if type(data) ~= "table" or data.version ~= 2 or type(data.settings) ~= "table" then return false end
    mergeSettings(data.settings)
    if type(data.binds) == "table" then
        local nextBinds, used = {}, {}
        -- Reserve the menu first; malformed duplicate bindings fall back to unbound.
        local menu = validKey(data.binds.menu) and data.binds.menu or "RightShift"
        nextBinds.menu, used[menu] = menu, true
        for _, key in ipairs({"killaura", "autoTotem", "camLock", "silentAim", "esp"}) do
            local value = data.binds[key]
            if validKey(value) and not used[value] then nextBinds[key], used[value] = value, true
            else nextBinds[key] = "None" end
        end
        for key, value in pairs(nextBinds) do binds[key] = value end
    end
    syncSilent(); resizePool(); applyTheme(); fitWindow(); refreshAll()
    return true
end
local PROFILE_FILE = "unstable-smp-v2.json"
local function saveProfile()
    local data = settingsSnapshot()
    env.UNSTABLE_SMP_PROFILE = data
    if type(writefile) == "function" then
        local ok = pcall(function() writefile(PROFILE_FILE, HttpService:JSONEncode(data)) end)
        report(ok and "Profile saved to unstable-smp-v2.json" or "Disk save failed; profile kept for this session.")
    else report("Profile saved for this session; file APIs unavailable.") end
end
local function restoreProfile()
    local data = env.UNSTABLE_SMP_PROFILE
    if type(readfile) == "function" then
        local ok, decoded = pcall(function()
            local raw = readfile(PROFILE_FILE)
            assert(#raw < 100000, "Profile too large")
            return HttpService:JSONDecode(raw)
        end)
        if ok then data = decoded end
    end
    report(loadProfile(data) and "Profile loaded." or "No compatible saved profile found.")
end

    end)
end

end

listen(UIS.InputBegan, function(input, gameProcessed)
    if not alive or typing or UIS:GetFocusedTextBox() then return end
    if capture then
        local name = input.KeyCode.Name
        if name == "Unknown" then return end
        if name == "Escape" then capture = nil; refreshAll(); return end
        if name == "Backspace" or name == "Delete" then
            if capture == "menu" then report("The menu needs a keybind.")
            else binds[capture] = "None"; capture = nil end
            refreshAll(); return
        end
        for key, value in pairs(binds) do
            if key ~= capture and value == name then report(name .. " is already bound to " .. bindNames[key] .. "."); return end
        end
        binds[capture] = name; capture = nil; refreshAll(); return
    end
    -- Menu remains reachable even if a game consumes its key, except while typing.
    if input.KeyCode.Name == binds.menu then setMenu(not menuOpen); return end
    if gameProcessed then return end
    if input.UserInputType == Enum.UserInputType.MouseButton2 then rightHeld = true end
    for key, value in pairs(binds) do
        if key ~= "menu" and value ~= "None" and input.KeyCode.Name == value then
            setOption(key, not CFG[key]); report(bindNames[key] .. (CFG[key] and " enabled" or " disabled")); break
        end
    end
end)

end)

local errors, errorTimes = {}, {}
local function guarded(name, fn, arg)
    local ok, message = pcall(fn, arg)
    if ok then errors[name] = 0; return end
    errors[name] = (errors[name] or 0) + 1
    local now = os.clock()
    if now - (errorTimes[name] or -10) > 5 then
        errorTimes[name] = now
        report(name .. ": " .. tostring(message))
        warn("UNSTABLE / " .. name .. ": " .. tostring(message))
    end
    if errors[name] >= 3 then
        if name == "Aimbot" then CFG.camLock = false; aimTarget = nil
        elseif name == "ESP" then CFG.esp = false; hideEsp()
        elseif name == "Killaura" then CFG.killaura = false
        elseif name == "Totem" then CFG.autoTotem = false; finishTransaction()
        elseif name == "Targets" then
            CFG.camLock, CFG.killaura, CFG.silentAim, CFG.esp = false, false, false, false
            aimTarget, auraTarget, silentTarget = nil, nil, nil
            syncSilent(); hideEsp()
        end
        refreshAll()
    end
end
local scanClock, espClock, statusClock, fpsClock, frames = 0, 0, 0, 0, 0
local displayFps = 0
listen(RunService.Heartbeat, function(dt)
    if not alive then return end
    local now = os.clock()
    scanClock, statusClock = scanClock + dt, statusClock + dt
    if scanClock >= 1 / CFG.scanHz then scanClock = 0; guarded("Targets", scanTargets, now) end
    if CFG.autoTotem or totTransaction then guarded("Totem", ensureTotem, now) end
    if CFG.killaura then guarded("Killaura", doKillaura, now) end
    if statusClock >= 0.25 then
        statusClock = 0
        local health = player:GetAttribute("Health")
        local hp = finite(health) and string.format("%.0f", health) or "?"
        if menuOpen then
            status.Text = string.format("HP %s   /   %d targets   /   scan %.2fms   /   ESP %.2fms   /   %d FPS", hp, count, scanCost, espCost, displayFps)
            fitWindow()
        end
        if CFG.watermark then markStat.Text = string.format("%d FPS / v2", displayFps) end
        if toast.Visible and now >= (toast:GetAttribute("Expires") or 0) then toast.Visible = false end
        if CFG.silentAim and wrappedSend and Net.send ~= wrappedSend then syncSilent(); refreshAll() end
    end
end)

listen(RunService.RenderStepped, function(dt)
    if not alive then return end
    fpsClock, frames, espClock = fpsClock + dt, frames + 1, espClock + dt
    if fpsClock >= 0.5 then displayFps = math.floor(frames / fpsClock + 0.5); fpsClock, frames = 0, 0 end
    if CFG.esp and espClock >= 1 / CFG.espHz then espClock = 0; guarded("ESP", drawEsp, os.clock()) end

    if fov.Visible then

        fov.Size = UDim2.fromOffset(CFG.aimFov * 2, CFG.aimFov * 2)
    end
end)
